## Fase 0 - Identificar a root
$ aws sts get-caller-identity --profile root
{
    "UserId": "000000000000",
    "Account": "000000000000",
    "Arn": "arn:aws:iam::000000000000:root"
}

$ aws iam list-users --profile root
{
    "Users": []
}

## Fase 1 - Usuario administrador
$ aws iam create-group --group-name Admins --profile root --query Group.Arn
"arn:aws:iam::000000000000:group/Admins"

$ aws iam attach-group-policy --group-name Admins --profile root --policy-arn arn:aws:iam::aws:policy/AdministratorAccess

$ aws iam create-user --user-name admin-lab --profile root --query User.Arn
"arn:aws:iam::000000000000:user/admin-lab"

$ aws iam add-user-to-group --group-name Admins --user-name admin-lab --profile root

$ aws sts get-caller-identity --profile admin
{
    "UserId": "AIDAFZEE84U0VGNWT37S",
    "Account": "000000000000",
    "Arn": "arn:aws:iam::000000000000:user/admin-lab"
}

## Fase 3.2 - Denegación explícita
$ aws iam add-user-to-group --group-name Sin-Descarga --user-name eva --profile admin

$ aws s3 cp s3://lab-datos/hola.txt descarga.txt --profile eva

$ aws iam remove-user-from-group --group-name Sin-Descarga --user-name eva --profile admin

## Fase 4 - MFA
Dispositivo: arn:aws:iam::000000000000:mfa/eva-mfa
Código 1: 406736
Código 2: 359146

$ aws iam enable-mfa-device --user-name eva --serial-number arn:aws:iam::000000000000:mfa/eva-mfa --authentication-code1 406736 --authentication-code2 359146 --profile admin

$ aws iam attach-group-policy --group-name EC2-Operadores --policy-arn arn:aws:iam::000000000000:policy/EC2-Requiere-MFA --profile admin
